 \section{Building The Graph}
 \begin{frame}[fragile]{Goal}
 From the given \verb|.tsv| file, build a \emph{Bipartite Unweighted Graph} where nodes are actors/actresses and movies. \\
 \pause \ \\ There is an edge between an actor/actress node A and movie node M if A participated in M.
 \end{frame}
 \begin{frame}{Nodes}
 Each node in the Graph contains specific attributes:
 \begin{itemize}
 \pause
     \item \textbf{Id}: to uniquely identify actors and movies within the graph
\pause
     \item \textbf{Name}: entity name extracted from file
\pause
     \item \textbf{Type}: actor or movie
\pause
     \item \textbf{Year}\ (\emph{only movie nodes}): indicates the release year
 \end{itemize}
     
 \end{frame}
 
\begin{frame}[fragile]{Supporting Structures}
The algorithm uses two dictionaries as supporting structures: \\
\begin{itemize}
\pause 
    \item Actor dictionary 
    \begin{verbatim}
        {actor_name : actor_id}
    \end{verbatim}
    \item Movie dictionary
    \begin{verbatim}
        {movie_name : movie_id}
    \end{verbatim}
\end{itemize}
\end{frame}

\section{Most prolific actors}
\begin{frame}{Goal}
Find the actor who did more movies, considering only the movies up to year x with
x in \{1930,1940,1950,1960,1970,1980,1990,2000,2010,2020\}.
\end{frame}
\begin{frame}[fragile]{Algorithm}
The algorithm examines every actor one by one and only once. \\
For each one, it counts the movies made considering an increasing year range, and stores the results in a dictionary
\begin{verbatim}
    {year: number of movies}
\end{verbatim}
The cumulative results are stored in a different dictionary
\begin{verbatim}
    {year: (number of movies, actor_id)}
\end{verbatim}
\end{frame}

\section{Diameter}
\begin{frame}{Goal}
Restricting to the largest connected component of the graph, compute exactly the diameter of the graph, considering only the movies up to year x with x in \{1930,1940,1950,1960,1970,1980,1990,2000,2010,2020\}
\end{frame}
\begin{frame}{Algorithm}
The algortithm implemented performs Breadth First Searches on the nodes in a specific order inferred by some properties of the BFS-trees, while at the same time: \\ \ \\
\begin{itemize}
    \item refining a lower bound (i.e. the maximum eccentricity found)
    \item upper bounding the remaining nodes' eccentricities
\end{itemize}
\ \\ \ \\
The algorithm stops when the lower bound is bigger than the upper bound.
\end{frame}
\begin{frame}{Algorithm}
\begin{itemize}
    \item Constructs a BFS-Tree starting from a high degree node \emph{source}
    \item Considering the BFS-Tree levels one at a time in a bottom-up fashion, compute the eccentricities of all nodes in the level
    \item If in a generic level $i$, the maximum eccentricity found $M$ is greater then $2*(i-1)$, the algorithm can stop.
\end{itemize}
\ \\ \ \\
It can be proved that all the eccentricities of the nodes in the upper layers of the tree are bounded by $max \{M,\ 2 \cdot(i-1)\}$
\end{frame}
\begin{frame}[fragile]{Code}
\begin{lstlisting}
layers, ecc = bfs_layers_ecc(graph, source)
i = ecc
lb = ecc
ub = 2 * ecc
while ub > lb:
    # compute B_i(source)
    F_i = layers[i]
    B_i = 0
    for node in F_i:
        _, ecc_i = bfs_layers_ecc(graph, source)
        if ecc_i > B_i:
            B_i = ecc_i

    if max(lb, B_i) > 2 * (i - 1): # lb > ub
        return max(lb, B_i)
    else:
        lb = max(lb, B_i)
        ub = 2 * (i - 1)
    i -= 1
return lb
\end{lstlisting}
\end{frame}

\section{Largest Shared Cast}
\begin{frame}{Goal}
Which is the pair of movies that share the largest number of actors?
\end{frame}
\begin{frame}{Algorithm}
The algorithm checks every movie pair combination possible and for each pair checks the number of common elements in their cast. \\
This last operation can be calculated as a set intersection. \\ \ \\
\pause
The total number of possible movie pair combinations is 
\begin{equation*}
    \binom{n}{2} = \ \frac{n!}{2\cdot(n-2)!} \ = \ \frac{n \ (n-1)}{2}
\end{equation*}
where n is the number of movies. \\ \ \\
The total number of comparisons can be much lower.
\end{frame}
\begin{frame}{Algorithm}
The current maximum provides a lower bound for the sought value
\pause
\begin{itemize}
    \item all the movies with a cast smaller than the current lower bound can be skipped
\end{itemize}
\end{frame}
\begin{frame}[fragile]{Code}
\begin{lstlisting}[basicstyle=\ssmall\ttfamily,]
# check movies in order
for i in range(len(movie_list) - 1):

  if movie_list[i] not in discarded:
  
    cast = set(imdb_graph.adj[movie_list[i]])
    
    if len(cast) > max_shared_cast:  
        
        # check the remaining movies 
        for j in range(i+1, len(movie_list)):
             
          if movie_list[j] not in discarded:
            other_cast = set(imdb_graph.adj[movie_list[j]])
            if len(other_cast) > max_shared_cast:  
            
              shared_cast = len(cast.intersection(other_cast))
              
              if shared_cast > max_shared_cast:
                max_shared_cast = shared_cast
                movie_pair = (movie_list[i], movie_list[j])
                
            else: # if j-th movie's cast is too small
              discarded.add(movie_list[j])
              
    else: # if i-th movie's cast is too small
        discarded.add(movie_list[i])

\section{Building The Actors' Graph}
\begin{frame}{Goal}
Build actor graph, whose nodes are only actors and two actors are connected if they did a movie together. \\ \ \\
Find the actors who collaborated the most among themselves.
\end{frame}
\begin{frame}{Algorithm}
The algorithm proposed creates a weighted graph where the weight of an edge represents how many times two actors have worked together. 
\\ \ \\
The edge with highest weight connects the actors who worked together the most.
    
\end{frame}
\begin{frame}[fragile]{Algorithm}
To build the graph, the algorithm relies on the previous actor-movie graph. \\ For each movie and its cast, consider all possible combinations of pairs of actors:
\begin{itemize}
    \item if actors' nodes already share an edge, increase its weight
    \item if not, create an edge with unitary weight
\end{itemize}
\pause
\end{frame}

\begin{frame}[fragile]{Code}
\begin{lstlisting}
# create a node for every actor
actors_graph.add_nodes_from(act_dict.values) 

# iterate over movies in the graph
for movie_id in movie_dict.values():  
  cast = list(graph.adj[movie_id])
  
  if len(cast) > 1:
    # consider all possible combinations
    for pair in itertools.combinations(cast, 2):
    
      if actors_graph.has_edge(pair[0], pair[1]):  
        # increase edge weight
        actors_graph.edges[pair[0], pair[1]]["weight"] += 1 
        
      else: 
        # add edge with unitary weight
        actors_graph.add_edge(pair[0], pair[1], weight=1)  

return actors_graph
\end{lstlisting}
\end{frame}
